   322→   322→function JobApplicationForm({ jobTitle, onClose }: { jobTitle: string; onClose: () => void }) {
   323→   323→  const { toast } = useToast();
   324→   324→  const [submitting, setSubmitting] = useState(false);
   325→   325→  const [submitted, setSubmitted] = useState(false);
   326→   326→  const [formStartTime] = useState(() => Date.now());
   327→   327→
   328→   328→  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
   329→   329→    e.preventDefault();
   330→   330→    setSubmitting(true);
   331→   331→
   332→   332→    const formData = new FormData(e.currentTarget);
   333→   333→    const email = formData.get("email") as string;
   334→   334→    const phone = formData.get("phone") as string;
   335→   335→    const botcheck = formData.get("botcheck");
   336→   336→
   337→   337→    if (botcheck) {
   338→   338→      setSubmitting(false);
   339→   339→      return;
   340→   340→    }
   341→   341→
   342→   342→    const elapsed = Date.now() - formStartTime;
   343→   343→    if (elapsed < 3000) {
   344→   344→      setSubmitting(false);
   345→   345→      toast({
   346→   346→        title: "Please take a moment",
   347→   347→        description: "Please review the form and try submitting again.",
   348→   348→      });
   349→   349→      return;
   350→   350→    }
   351→   351→
   352→   352→    const applicationKey = `landmark_application`;
   353→   353→    const existingApplication = localStorage.getItem(applicationKey);
   354→   354→    if (existingApplication) {
   355→   355→      try {
   356→   356→        const app = JSON.parse(existingApplication);
   357→   357→        if (app.email === email || app.phone === phone) {
   358→   358→          setSubmitting(false);
   359→   359→          setSubmitted(true);
   360→   360→          toast({
   361→   361→            title: "Already Applied!",
   362→   362→            description: `You have already submitted an application for "${app.jobTitle}". Our team will contact you soon.`,
   363→   363→          });
   364→   364→          setTimeout(onClose, 5000);
   365→   365→          return;
   366→   366→        }
   367→   367→      } catch {
   368→   368→        // ignore
   369→   369→      }
   370→   370→    }
   371→   371→
   372→   372→    const payload = {
   373→   373→      name: formData.get("fullName"),
   374→   374→      email,
   375→   375→      phone,
   376→   376→      nationality: formData.get("nationality"),
   377→   377→      currentLocation: formData.get("currentLocation"),
   378→   378→      experience: formData.get("experience"),
   379→   379→      certifications: formData.get("certifications"),
   380→   380→      languages: formData.get("languages"),
   381→   381→      iqamaStatus: formData.get("iqamaStatus"),
   382→   382→      noticePeriod: formData.get("noticePeriod"),
   383→   383→      message: formData.get("message"),
   384→   384→      jobTitle,
   385→   385→      botcheck,
   386→   386→      formStartTime,
   387→   387→    };
   388→   388→
   389→   389→    try {
   390→   390→      const response = await fetch("/api/job-application", {
   391→   391→        method: "POST",
   392→   392→        headers: {
   393→   393→          "Content-Type": "application/json",
   394→   394→          Accept: "application/json",
   395→   395→        },
   396→   396→        body: JSON.stringify(payload),
   397→   397→      });
   398→   398→
   399→   399→      const result = await response.json();
   400→   400→
   401→   401→      if (response.ok && result.success) {
   402→   402→        localStorage.setItem(applicationKey, JSON.stringify({ email, phone, jobTitle, date: new Date().toISOString() }));
   403→   403→        setSubmitted(true);
   404→   404→        toast({
   405→   405→          title: "Application Submitted!",
   406→   406→          description: `We'll review your application for "${jobTitle}" and contact you soon.`,
   407→   407→        });
   408→   408→        (e.target as HTMLFormElement).reset();
   409→   409→        setTimeout(onClose, 3000);
   410→   410→      } else {
   411→   411→        setSubmitting(false);
   412→   412→        toast({
   413→   413→          title: "Submission failed",
   414→   414→          description: result.message || "Please try again later.",
   415→   415→          variant: "destructive",
   416→   416→        });
   417→   417→      }
   418→   418→    } catch {
   419→   419→      setSubmitting(false);
   420→   420→      toast({
   421→   421→        title: "Network error",
   422→   422→        description: "Please check your internet connection and try again.",
   423→   423→        variant: "destructive",
   424→   424→      });
   425→   425→    } finally {
   426→   426→      setSubmitting(false);
   427→   427→    }
   428→   428→  };
   429→   429→
   430→   430→  return (
   431→   431→    <motion.div
   432→   432→      initial={{ opacity: 0 }}
   433→   433→      animate={{ opacity: 1 }}
   434→   434→      exit={{ opacity: 0 }}
   435→   435→      className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-primary/80 backdrop-blur-sm"
   436→   436→      onClick={onClose}
   437→   437→    >
   438→   438→      <motion.div
   439→   439→        initial={{ scale: 0.9, y: 20 }}
   440→   440→        animate={{ scale: 1, y: 0 }}
   441→   441→        exit={{ scale: 0.9, y: 20 }}
   442→   442→        transition={{ type: "spring", stiffness: 300, damping: 30 }}
   443→   443→        className="w-full max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl bg-background border-2 border-accent/30 shadow-2xl"
   444→   444→        onClick={(e) => e.stopPropagation()}
   445→   445→      >
   446→   446→        {submitted ? (
   447→   447→          <div className="relative p-8 text-center overflow-hidden">
   448→   448→            <div className="absolute inset-0 pointer-events-none" aria-hidden="true">
   449→   449→              {[...Array(14)].map((_, i) => (
   450→   450→                <motion.span
   451→   451→                  key={i}
   452→   452→                  initial={{ opacity: 0, y: -20, x: 0 }}
