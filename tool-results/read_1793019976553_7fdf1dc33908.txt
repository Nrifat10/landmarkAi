   322→function JobApplicationForm({ jobTitle, onClose }: { jobTitle: string; onClose: () => void }) {
   323→  const { toast } = useToast();
   324→  const [submitting, setSubmitting] = useState(false);
   325→  const [submitted, setSubmitted] = useState(false);
   326→  const [formStartTime] = useState(() => Date.now());
   327→
   328→  const handleSubmit = async (e: React.FormEvent<HTMLFormElement>) => {
   329→    e.preventDefault();
   330→    setSubmitting(true);
   331→
   332→    const formData = new FormData(e.currentTarget);
   333→    const email = formData.get("email") as string;
   334→    const phone = formData.get("phone") as string;
   335→    const botcheck = formData.get("botcheck");
   336→
   337→    if (botcheck) {
   338→      setSubmitting(false);
   339→      return;
   340→    }
   341→
   342→    const elapsed = Date.now() - formStartTime;
   343→    if (elapsed < 3000) {
   344→      setSubmitting(false);
   345→      toast({
   346→        title: "Please take a moment",
   347→        description: "Please review the form and try submitting again.",
   348→      });
   349→      return;
   350→    }
   351→
   352→    const applicationKey = `landmark_application`;
   353→    const existingApplication = localStorage.getItem(applicationKey);
   354→    if (existingApplication) {
   355→      try {
   356→        const app = JSON.parse(existingApplication);
   357→        if (app.email === email || app.phone === phone) {
   358→          setSubmitting(false);
   359→          setSubmitted(true);
   360→          toast({
   361→            title: "Already Applied!",
   362→            description: `You have already submitted an application for "${app.jobTitle}". Our team will contact you soon.`,
   363→          });
   364→          setTimeout(onClose, 5000);
   365→          return;
   366→        }
   367→      } catch {
   368→        // ignore
   369→      }
   370→    }
   371→
   372→    const payload = {
   373→      name: formData.get("fullName"),
   374→      email,
   375→      phone,
   376→      nationality: formData.get("nationality"),
   377→      currentLocation: formData.get("currentLocation"),
   378→      experience: formData.get("experience"),
   379→      certifications: formData.get("certifications"),
   380→      languages: formData.get("languages"),
   381→      iqamaStatus: formData.get("iqamaStatus"),
   382→      noticePeriod: formData.get("noticePeriod"),
   383→      message: formData.get("message"),
   384→      jobTitle,
   385→      botcheck,
   386→      formStartTime,
   387→    };
   388→
   389→    try {
   390→      const response = await fetch("/api/job-application", {
   391→        method: "POST",
   392→        headers: {
   393→          "Content-Type": "application/json",
   394→          Accept: "application/json",
   395→        },
   396→        body: JSON.stringify(payload),
   397→      });
   398→
   399→      const result = await response.json();
   400→
   401→      if (response.ok && result.success) {
   402→        localStorage.setItem(applicationKey, JSON.stringify({ email, phone, jobTitle, date: new Date().toISOString() }));
   403→        setSubmitted(true);
   404→        toast({
   405→          title: "Application Submitted!",
   406→          description: `We'll review your application for "${jobTitle}" and contact you soon.`,
   407→        });
   408→        (e.target as HTMLFormElement).reset();
   409→        setTimeout(onClose, 3000);
   410→      } else {
   411→        setSubmitting(false);
   412→        toast({
   413→          title: "Submission failed",
   414→          description: result.message || "Please try again later.",
   415→          variant: "destructive",
   416→        });
   417→      }
   418→    } catch {
   419→      setSubmitting(false);
   420→      toast({
   421→        title: "Network error",
   422→        description: "Please check your internet connection and try again.",
   423→        variant: "destructive",
   424→      });
   425→    } finally {
   426→      setSubmitting(false);
   427→    }
   428→  };
   429→
   430→  return (
   431→    <motion.div
   432→      initial={{ opacity: 0 }}
   433→      animate={{ opacity: 1 }}
   434→      exit={{ opacity: 0 }}
   435→      className="fixed inset-0 z-[100] flex items-center justify-center p-4 bg-primary/80 backdrop-blur-sm"
   436→      onClick={onClose}
   437→    >
   438→      <motion.div
   439→        initial={{ scale: 0.9, y: 20 }}
   440→        animate={{ scale: 1, y: 0 }}
   441→        exit={{ scale: 0.9, y: 20 }}
   442→        transition={{ type: "spring", stiffness: 300, damping: 30 }}
   443→        className="w-full max-w-lg max-h-[90vh] overflow-y-auto rounded-2xl bg-background border-2 border-accent/30 shadow-2xl"
   444→        onClick={(e) => e.stopPropagation()}
   445→      >
   446→        {submitted ? (
   447→          <div className="relative p-8 text-center overflow-hidden">
   448→            <div className="absolute inset-0 pointer-events-none" aria-hidden="true">
   449→              {[...Array(14)].map((_, i) => (
   450→                <motion.span
   451→                  key={i}
   452→                  initial={{ opacity: 0, y: -20, x: 0 }}
   453→                  animate={{
   454→                    opacity: [0, 1, 0],
   455→                    y: [0, 280],
   456→                    x: [(i - 7) * 18, (i - 7) * 30],
   457→                    rotate: [0, 360],
   458→                  }}
   459→                  transition={{ duration: 1.8, delay: i * 0.08, ease: "easeOut" }}
   460→                  className="absolute top-2 left-1/2 h-2 w-2 rounded-full"
   461→                  style={{
   462→                    backgroundColor: ["#f59e0b", "#10b981", "#3b82f6", "#ef4444", "#8b5cf6"][i % 5],
   463→                  }}
   464→                />
   465→              ))}
   466→            </div>
   467→
   468→            <motion.div
   469→              initial={{ scale: 0, rotate: -30 }}
   470→              animate={{ scale: 1, rotate: 0 }}
   471→              transition={{ type: "spring", stiffness: 200, damping: 12 }}
   472→              className="mx-auto mb-4 flex h-20 w-20 items-center justify-center rounded-full bg-green-500/10 ring-4 ring-green-500/20"
   473→            >
   474→              <motion.div
   475→                initial={{ scale: 0 }}
   476→                animate={{ scale: 1 }}
   477→                transition={{ delay: 0.2, type: "spring", stiffness: 300, damping: 15 }}
   478→              >
   479→                <CheckCircle2 className="h-11 w-11 text-green-500" strokeWidth={2.5} />
   480→              </motion.div>
   481→            </motion.div>
   482→
   483→            <motion.div
   484→              initial={{ opacity: 0, y: 10 }}
   485→              animate={{ opacity: 1, y: 0 }}
   486→              transition={{ delay: 0.4 }}
   487→            >
   488→              <div className="flex items-center justify-center gap-2 mb-2">
   489→                <PartyPopper className="h-5 w-5 text-accent" />
   490→                <h3 className="text-xl font-extrabold text-foreground">Application Submitted!</h3>
   491→              </div>
   492→              <p className="mt-1 text-sm text-muted-foreground">
   493→                Thank you for applying for <strong className="text-foreground">{jobTitle}</strong>. Our HR team will review your application and contact you within <strong>2–5 working days</strong>.
   494→              </p>

   496→              <div className="mt-5 rounded-xl border-2 border-accent/20 bg-accent/5 p-4 text-left">
